useEffect - Side Effects

import { useEffect, useState } from 'react';

function UserProfile({ userId }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Runs after component mounts and when userId changes
    async function fetchUser() {
      setLoading(true);
      const response = await fetch(`/api/users/${userId}`);
      const data = await response.json();
      setUser(data);
      setLoading(false);
    }

    fetchUser();

    // Cleanup function (optional)
    return () => {
      // Cancel requests, clear timers, etc.
    };
  }, [userId]); // Dependency array

  if (loading) return <div>Loading...</div>;
  if (!user) return <div>User not found</div>;

  return <div>{user.name}</div>;
}

// Common useEffect patterns
useEffect(() => {
  // Runs ONCE on mount (empty deps)
}, []);

useEffect(() => {
  // Runs on EVERY render (no deps array - avoid this!)
});

useEffect(() => {
  // Runs when count changes
}, [count]);

useContext - Global State

import { createContext, useContext, useState } from 'react';

// Create context
const AuthContext = createContext();

// Provider component
function AuthProvider({ children }) {
  const [user, setUser] = useState(null);

  const login = async (email, password) => {
    const response = await fetch('/api/login', {
      method: 'POST',
      body: JSON.stringify({ email, password })
    });
    const data = await response.json();
    setUser(data.user);
  };

  const logout = () => setUser(null);

  return (
    <AuthContext.Provider value={{ user, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

// Custom hook to use auth
function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within AuthProvider');
  }
  return context;
}

// Usage in components
function Dashboard() {
  const { user, logout } = useAuth();

  return (
    <div>
      <h1>Welcome, {user.name}</h1>
      <button onClick={logout}>Logout</button>
    </div>
  );
}

// App setup
function App() {
  return (
    <AuthProvider>
      <Dashboard />
    </AuthProvider>
  );
}

Custom Hooks

// useFetch - Reusable data fetching
function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    async function fetchData() {
      try {
        setLoading(true);
        const response = await fetch(url);
        const json = await response.json();
        setData(json);
      } catch (err) {
        setError(err);
      } finally {
        setLoading(false);
      }
    }

    fetchData();
  }, [url]);

  return { data, loading, error };
}

// Usage
function UserList() {
  const { data: users, loading, error } = useFetch('/api/users');

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}

// useLocalStorage - Persist state
function useLocalStorage(key, initialValue) {
  const [value, setValue] = useState(() => {
    const item = localStorage.getItem(key);
    return item ? JSON.parse(item) : initialValue;
  });

  useEffect(() => {
    localStorage.setItem(key, JSON.stringify(value));
  }, [key, value]);

  return [value, setValue];
}

// Usage
function Settings() {
  const [theme, setTheme] = useLocalStorage('theme', 'light');

  return (
    <button onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}>
      Current theme: {theme}
    </button>
  );
}

React Query / TanStack Query

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

// Fetch data with caching
function UserList() {
  const { data, isLoading, error } = useQuery({
    queryKey: ['users'],
    queryFn: () => fetch('/api/users').then(res => res.json())
  });

  if (isLoading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;

  return (
    <ul>
      {data.map(user => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}

// Mutations (create/update/delete)
function CreateUser() {
  const queryClient = useQueryClient();

  const mutation = useMutation({
    mutationFn: (newUser) => 
      fetch('/api/users', {
        method: 'POST',
        body: JSON.stringify(newUser)
      }),
    onSuccess: () => {
      // Invalidate and refetch
      queryClient.invalidateQueries({ queryKey: ['users'] });
    }
  });

  const handleSubmit = (e) => {
    e.preventDefault();
    mutation.mutate({ name: 'John', email: 'john@example.com' });
  };

  return (
    <form onSubmit={handleSubmit}>
      <button type="submit" disabled={mutation.isPending}>
        {mutation.isPending ? 'Creating...' : 'Create User'}
      </button>
      {mutation.isError && <div>Error: {mutation.error.message}</div>}
    </form>
  );
}

Real-Time with WebSockets

function ChatRoom({ roomId }) {
  const [messages, setMessages] = useState([]);
  const [ws, setWs] = useState(null);

  useEffect(() => {
    // Connect
    const websocket = new WebSocket(`ws://localhost:8000/ws/${roomId}`);

    websocket.onmessage = (event) => {
      const message = JSON.parse(event.data);
      setMessages(prev => [...prev, message]);
    };

    setWs(websocket);

    // Cleanup
    return () => websocket.close();
  }, [roomId]);

  const sendMessage = (text) => {
    if (ws) {
      ws.send(JSON.stringify({ text, timestamp: Date.now() }));
    }
  };

  return (
    <div>
      {messages.map((msg, i) => (
        <div key={i}>{msg.text}</div>
      ))}
      <MessageInput onSend={sendMessage} />
    </div>
  );
}

Performance Optimization

useMemo - Expensive Calculations

import { useMemo } from 'react';

function DataTable({ items, filter }) {
  // Only recalculate when items or filter changes
  const filteredItems = useMemo(() => {
    console.log('Filtering...'); // Only logs when deps change
    return items.filter(item => 
      item.name.toLowerCase().includes(filter.toLowerCase())
    );
  }, [items, filter]);

  return (
    <table>
      {filteredItems.map(item => (
        <tr key={item.id}>
          <td>{item.name}</td>
        </tr>
      ))}
    </table>
  );
}

useCallback - Memoized Functions

import { useCallback } from 'react';

function Parent() {
  const [count, setCount] = useState(0);

  // Without useCallback: new function every render
  // Child re-renders unnecessarily

  // With useCallback: same function reference
  const handleClick = useCallback(() => {
    console.log('Clicked');
  }, []); // Empty deps = never changes

  return (
    <>
      <p>{count}</p>
      <button onClick={() => setCount(count + 1)}>Increment</button>
      <Child onClick={handleClick} />
    </>
  );
}

// Child only re-renders when onClick changes
const Child = React.memo(({ onClick }) => {
  console.log('Child render');
  return <button onClick={onClick}>Click me</button>;
});

React.memo - Prevent Re-renders

// Without React.memo: re-renders every time parent re-renders
function UserCard({ user }) {
  console.log('Rendering UserCard');
  return <div>{user.name}</div>;
}

// With React.memo: only re-renders when user prop changes
const UserCard = React.memo(({ user }) => {
  console.log('Rendering UserCard');
  return <div>{user.name}</div>;
});

// Custom comparison
const UserCard = React.memo(
  ({ user }) => <div>{user.name}</div>,
  (prevProps, nextProps) => {
    // Return true if props are equal (skip re-render)
    return prevProps.user.id === nextProps.user.id;
  }
);

Interview Questions & Answers

Q1: "Explain the difference between useState and useRef."

Your Answer:

"Both store values, but behave differently:

useState:
- For values that affect rendering
- Updating triggers re-render
- Use for UI state

jsx const [count, setCount] = useState(0); setCount(1); // Component re-renders

useRef:
- For values that DON'T affect rendering
- Updating does NOT trigger re-render
- Persists across renders
- Use for: DOM refs, timers, previous values

jsx const countRef = useRef(0); countRef.current = 1; // No re-render

Real Example:
Form with auto-save. Use useState for input value (renders as user types). Use useRef for debounce timer (doesn't need to render):

```jsx
const [text, setText] = useState('');
const timerRef = useRef(null);

const handleChange = (e) => {
setText(e.target.value);

clearTimeout(timerRef.current);
timerRef.current = setTimeout(() => {
saveToServer(e.target.value);
}, 1000);
};
```"*

Q2: "How do you prevent unnecessary re-renders in React?"

Your Answer:

"Multiple techniques depending on the scenario:

1. React.memo (Component Level)
Memoize entire component:
jsx const UserCard = React.memo(({ user }) => { return <div>{user.name}</div>; });

2. useMemo (Value Level)
Memoize expensive calculations:
jsx const sortedUsers = useMemo(() => { return users.sort((a, b) => a.name.localeCompare(b.name)); }, [users]);

3. useCallback (Function Level)
Memoize functions passed as props:
jsx const handleClick = useCallback(() => { console.log('clicked'); }, []);

4. Proper Key Props
```jsx
// ❌ BAD: index as key

// ✅ GOOD: stable unique id
{items.map(item => )}
```

5. Code Splitting
Load components only when needed:
jsx const Dashboard = lazy(() => import('./Dashboard'));

Real Example:
Data table with 1000 rows. Without optimization: 2-second re-render on every keystroke. With React.memo on table rows + useMemo for filtered data: 50ms."

Q3: "How do you integrate a React frontend with a FastAPI backend?"

Your Answer:

"Here's my production setup:

Architecture:
React (localhost:3000) ←→ FastAPI (localhost:8000)

1. CORS Setup (FastAPI)
```python
from fastapi.middleware.cors import CORSMiddleware

app.add_middleware(
CORSMiddleware,
allow_origins=["http://localhost:3000"],
allow_credentials=True,
allow_methods=[""],
allow_headers=[""]
)
```

2. API Client (React)
```jsx
const API_URL = process.env.REACT_APP_API_URL || 'http://localhost:8000';

async function fetchUsers() {
const response = await fetch(${API_URL}/api/users, {
headers: {
'Authorization': Bearer ${token},
'Content-Type': 'application/json'
}
});
return response.json();
}
```

3. Custom Hook for API Calls
```jsx
function useAPI(endpoint) {
const [data, setData] = useState(null);
const [loading, setLoading] = useState(true);
const [error, setError] = useState(null);

useEffect(() => {
fetch(${API_URL}${endpoint})
.then(res => res.json())
.then(setData)
.catch(setError)
.finally(() => setLoading(false));
}, [endpoint]);

return { data, loading, error };
}
```

4. React Query (Better)
jsx const { data: users } = useQuery({ queryKey: ['users'], queryFn: () => fetch('/api/users').then(r => r.json()) });

5. WebSocket for Real-Time
jsx useEffect(() => { const ws = new WebSocket('ws://localhost:8000/ws'); ws.onmessage = (event) => { setMessages(prev => [...prev, JSON.parse(event.data)]); }; return () => ws.close(); }, []);

Production:
- Same origin (deploy React build via FastAPI static files)
- Or separate: React on Vercel, FastAPI on AWS
- API key / JWT for auth
- Rate limiting on backend"*
